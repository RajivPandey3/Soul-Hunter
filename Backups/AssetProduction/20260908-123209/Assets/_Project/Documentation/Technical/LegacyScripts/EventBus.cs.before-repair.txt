using System;
using System.Collections.Generic;
using SoulHunter.Core.Services;

namespace SoulHunter.Core.Events
{
    /// <summary>
    /// Learning Comment:
    /// Architecture Document (event-bus.md) ke mutabiq ye EventBus ki canonical implementation hai.
    /// Ye ek Service ban chuka hai jo IGameService implement karta hai (no global state).
    /// Sirf Subscribe, Unsubscribe, aur Raise methods hain (Hierarchical Single Responsibility).
    /// </summary>
    public class EventBus : IGameService
    {
        private readonly Dictionary<Type, Delegate> _subscribers = new Dictionary<Type, Delegate>();

        public void Initialize()
        {
            // Required by IGameService. Currently nothing to initialize.
        }

        /// <summary>
        /// Kisi event ke liye listener add karna.
        /// </summary>
        public void Subscribe<T>(Action<T> handler) where T : IGameEvent
        {
            var type = typeof(T);
            if (!_subscribers.ContainsKey(type))
            {
                _subscribers[type] = handler;
            }
            else
            {
                _subscribers[type] = Delegate.Combine(_subscribers[type], handler);
            }
        }

        /// <summary>
        /// Listener ko hatana taaki object destroy hone par memory leak na ho.
        /// </summary>
        public void Unsubscribe<T>(Action<T> handler) where T : IGameEvent
        {
            var type = typeof(T);
            if (_subscribers.TryGetValue(type, out var currentDel))
            {
                _subscribers[type] = Delegate.Remove(currentDel, handler);
                
                if (_subscribers[type] == null)
                {
                    _subscribers.Remove(type);
                }
            }
        }

        /// <summary>
        /// Event ko fire/broadcast karna (Architecture mein Raise method define tha).
        /// </summary>
        public void Raise<T>(T eventData) where T : IGameEvent
        {
            var type = typeof(T);
            if (_subscribers.TryGetValue(type, out var currentDel))
            {
                var action = currentDel as Action<T>;
                action?.Invoke(eventData);
            }
        }
        
        /// <summary>
        /// Game shutdown ya scene change ke waqt sab clear karna.
        /// </summary>
        public void Clear()
        {
            _subscribers.Clear();
        }
    }
}
