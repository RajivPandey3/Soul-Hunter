using UnityEngine;
using System.Collections.Generic;
using SoulHunter.Gameplay.Data;

namespace SoulHunter.Gameplay.AI
{
    /// <summary>
    /// Learning Comment:
    /// Kami #1 Fix: Ab ye spawner WaveData ke hisaab se kaam karega.
    /// Time badhne par naye dushman aur bosses aayenge.
    /// Kyunki alag-alag dushman aayenge, humein Object Pool ko Dictionary mein badalna pada.
    /// </summary>
    public class EnemySpawner : MonoBehaviour
    {
        [SerializeField] private List<WaveData> _waves;
        [SerializeField] private float _spawnRadius = 20f;
        
        private Transform _playerTransform;
        private float _timer;
        private WaveData _currentWave;
        private bool _bossSpawnedForCurrentWave;

        // Har prefab ke liye alag Object Pool (List)
        private Dictionary<GameObject, List<GameObject>> _enemyPools = new Dictionary<GameObject, List<GameObject>>();

        private void Start()
        {
            var player = GameObject.FindWithTag("Player");
            if (player != null)
            {
                _playerTransform = player.transform;
            }

            // Waves ko time ke hisaab se sort kar lo taaki koi gadbad na ho
            if (_waves != null && _waves.Count > 0)
            {
                _waves.Sort((a, b) => a.StartTimeInSeconds.CompareTo(b.StartTimeInSeconds));
            }
        }

        private void Update()
        {
            if (_playerTransform == null || _waves == null || _waves.Count == 0) return;

            float timePassed = 0f;
            if (SoulHunter.Gameplay.Core.GameSessionManager.Instance != null)
            {
                timePassed = SoulHunter.Gameplay.Core.GameSessionManager.Instance.SurvivalTime;
            }

            // Check karo ki abhi kaunsi wave chalni chahiye
            WaveData activeWave = _waves[0];
            for (int i = 0; i < _waves.Count; i++)
            {
                if (timePassed >= _waves[i].StartTimeInSeconds)
                {
                    activeWave = _waves[i];
                }
            }

            // Agar wave badal gayi (jaise 0s se 60s par aa gaye)
            if (_currentWave != activeWave)
            {
                _currentWave = activeWave;
                _bossSpawnedForCurrentWave = false;
            }

            if (_currentWave == null || _currentWave.EnemyPrefab == null) return;

            // Agar ye boss wave hai aur boss aa chuka hai, toh ruk jao
            if (_currentWave.IsBossWave && _bossSpawnedForCurrentWave) return;

            _timer -= Time.deltaTime;
            if (_timer <= 0f)
            {
                for (int i = 0; i < _currentWave.EnemiesPerSpawn; i++)
                {
                    SpawnEnemy(_currentWave.EnemyPrefab);
                }

                if (_currentWave.IsBossWave)
                {
                    _bossSpawnedForCurrentWave = true;
                }

                _timer = _currentWave.SpawnInterval;
            }
        }

        private void SpawnEnemy(GameObject prefab)
        {
            Vector2 randomDir = Random.insideUnitCircle.normalized;
            float randomDist = Random.Range(_spawnRadius, _spawnRadius + 10f);
            
            Vector3 spawnPos = _playerTransform.position + new Vector3(randomDir.x * randomDist, 0f, randomDir.y * randomDist);
            spawnPos.y = _playerTransform.position.y; 

            // Initialize the pool list for this specific prefab if it doesn't exist
            if (!_enemyPools.ContainsKey(prefab))
            {
                _enemyPools[prefab] = new List<GameObject>();
            }

            List<GameObject> pool = _enemyPools[prefab];
            GameObject enemyToSpawn = null;
            
            for (int i = 0; i < pool.Count; i++)
            {
                if (!pool[i].activeInHierarchy)
                {
                    enemyToSpawn = pool[i];
                    break;
                }
            }

            if (enemyToSpawn == null)
            {
                enemyToSpawn = Instantiate(prefab, spawnPos, Quaternion.identity);
                pool.Add(enemyToSpawn);
            }

            enemyToSpawn.transform.position = spawnPos;
            enemyToSpawn.SetActive(true);
            
            var health = enemyToSpawn.GetComponent<SoulHunter.Gameplay.Combat.HealthController>();
            if (health != null)
            {
                health.ResetHealth();
            }
        }
    }
}
