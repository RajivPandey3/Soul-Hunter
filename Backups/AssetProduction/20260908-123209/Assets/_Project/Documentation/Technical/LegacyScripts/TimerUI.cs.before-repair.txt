using UnityEngine;
using TMPro;
using SoulHunter.Gameplay.Core;

namespace SoulHunter.Gameplay.UI
{
    /// <summary>
    /// Learning Comment:
    /// Screen ke upar time dikhane ke liye script.
    /// Ye GameSessionManager se har frame (ya har second) naya time le kar UI ko update karti hai.
    /// </summary>
    public class TimerUI : MonoBehaviour
    {
        [SerializeField] private TextMeshProUGUI _timeText;

        private void Update()
        {
            if (GameSessionManager.Instance != null && _timeText != null)
            {
                _timeText.text = GameSessionManager.Instance.GetFormattedTime();
            }
        }
    }
}
