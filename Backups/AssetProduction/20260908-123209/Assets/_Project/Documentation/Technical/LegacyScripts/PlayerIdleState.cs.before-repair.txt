using UnityEngine;

namespace SoulHunter.Gameplay.Player
{
    public class PlayerIdleState : IPlayerState
    {
        private readonly PlayerController _controller;

        public PlayerIdleState(PlayerController controller)
        {
            _controller = controller;
        }

        public void Enter()
        {
            // Future: Play Idle Animation
        }

        public void UpdateLogic()
        {
            if (_controller.CurrentMoveInput.sqrMagnitude > 0.01f)
            {
                _controller.ChangeState(new PlayerRunState(_controller));
            }
        }

        public void UpdatePhysics(ref SoulHunter.Gameplay.Physics.EnvironmentData envData)
        {
            // Halt any residual movement smoothly
            _controller.Rigidbody.linearVelocity = new Vector3(0, _controller.Rigidbody.linearVelocity.y, 0);

            if (_controller.Animator != null)
            {
                _controller.Animator.UpdateSpeed(0f);
            }
        }

        public void Exit()
        {
        }
    }
}
