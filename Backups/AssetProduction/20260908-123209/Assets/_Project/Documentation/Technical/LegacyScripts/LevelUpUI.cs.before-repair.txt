using UnityEngine;
using UnityEngine.UI;
using TMPro;
using System.Collections.Generic;
using SoulHunter.Gameplay.Player;
using SoulHunter.Gameplay.Combat;

namespace SoulHunter.Gameplay.UI
{
    /// <summary>
    /// Learning Comment:
    /// VS = SH Rule: Level up par game pause hota hai aur 3 random cards aate hain.
    /// Ye script buttons par random upgrades set karti hai aur click hone par WeaponManager ko bhejti hai.
    /// </summary>
    public class LevelUpUI : MonoBehaviour
    {
        [SerializeField] private PlayerExperience _playerExperience;
        [SerializeField] private WeaponManager _weaponManager;
        [SerializeField] private GameObject _levelUpPanel;
        [SerializeField] private Button[] _upgradeButtons;
        
        [Tooltip("Sare banaye hue upgrade cards yahan daalein")]
        [SerializeField] private List<UpgradeData> _allUpgrades;

        private void OnEnable()
        {
            if (_playerExperience != null)
            {
                _playerExperience.OnLevelUp += HandleLevelUp;
            }
            
            if (_levelUpPanel != null)
            {
                _levelUpPanel.SetActive(false);
            }
        }

        private void OnDisable()
        {
            if (_playerExperience != null)
            {
                _playerExperience.OnLevelUp -= HandleLevelUp;
            }
        }

        private void HandleLevelUp(int newLevel)
        {
            ShowLevelUpScreen(newLevel);
        }

        private void ShowLevelUpScreen(int newLevel)
        {
            if (SoulHunter.Gameplay.Audio.AudioManager.Instance != null)
            {
                // Unscaled time play is not directly supported by basic AudioSource without changing pitch,
                // but the AudioManager plays it and Time.timeScale=0 doesn't stop sounds unless paused explicitly.
                SoulHunter.Gameplay.Audio.AudioManager.Instance.PlaySFX(SoulHunter.Gameplay.Audio.AudioManager.Instance.LevelUpSound);
            }

            Time.timeScale = 0f;
            
            if (_allUpgrades.Count > 0)
            {
                // Teeno buttons par random upgrade set karo
                for (int i = 0; i < _upgradeButtons.Length; i++)
                {
                    if (_upgradeButtons[i] != null)
                    {
                        // Abhi hum simply random le rahe hain. (Asli VS me thoda smart random hota hai)
                        UpgradeData randomUpgrade = _allUpgrades[Random.Range(0, _allUpgrades.Count)];
                        
                        // Button ka text change karo (TextMeshPro required)
                        TextMeshProUGUI btnText = _upgradeButtons[i].GetComponentInChildren<TextMeshProUGUI>();
                        if (btnText != null)
                        {
                            btnText.text = randomUpgrade.UpgradeName;
                        }

                        // Purane clicks hatao aur naya click event lagao
                        _upgradeButtons[i].onClick.RemoveAllListeners();
                        _upgradeButtons[i].onClick.AddListener(() => OnUpgradeSelected(randomUpgrade));
                    }
                }
            }

            if (_levelUpPanel != null)
            {
                _levelUpPanel.SetActive(true);
            }
            
            Debug.Log($"[LevelUpUI] Game Paused for Level {newLevel} Upgrades!");
        }

        private void OnUpgradeSelected(UpgradeData selectedUpgrade)
        {
            Debug.Log($"[LevelUpUI] Player Selected: {selectedUpgrade.UpgradeName}");
            
            if (_weaponManager != null)
            {
                _weaponManager.ApplyUpgrade(selectedUpgrade);
            }

            if (_levelUpPanel != null)
            {
                _levelUpPanel.SetActive(false);
            }
            
            Time.timeScale = 1f;
        }
    }
}
